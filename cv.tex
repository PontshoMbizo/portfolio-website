\section{PROJECTS}

\workentry{Weeding Gantry Robot --- final-year research project (EEE4022S)}{2026}
\begin{itemize}
  \item Built a low-cost three-axis gantry (Raspberry Pi Zero W, ESP32-C3) that photographs a crop bed, classifies weeds on the device and sprays each one; wrote the Python operator software and the ESP32 closed-loop motion firmware with homing, stall detection and a safety watchdog.
  \item 245 closed-loop moves ended within one sensor count (1.07 mm); worst settling time 1.86 s against a 3 s limit; positioning within 14 mm against a 20 mm requirement. Spray-grid detection F1 0.908 for segmentation against 0.654 for YOLOv8n.
\end{itemize}
\vspace{2pt}
\workentry{fpgasim and mcusim --- in-browser hardware simulators}{2025 -- 2026}
\begin{itemize}
  \item \textbf{fpgasim:} Verilog in the browser, simulated with Icarus Verilog in ephemeral, network-less containers, on a Tang Nano 9K board model with waveforms; 69 unit tests, 272--375 ms round trip.
  \item \textbf{mcusim:} bare-metal C for an STM32F103 Blue Pill, compiled with arm-none-eabi-gcc and run in QEMU, with the GPIO trace drawn on a board view; 68 unit tests, 357--469 ms round trip.
\end{itemize}
\vspace{2pt}
\workentry{scope.io --- interview-prep platform for electronics engineers}{2026}

\workentry{fulgurlab --- Simulink-style block-diagram simulator}{2026}
